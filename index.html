<script>

function validateForm() {

const form = document.getElementById('palpitesForm');

const errorMessage = document.getElementById('errorMessage');

let isValid = true;

for (let i = 1; i <= 10; i++) {

const game = form.querySelector(`input[name="game${i}"]:checked`);

if (!game) {

isValid = false;

break;

}

}

if (!isValid) {

errorMessage.style.display = 'block';

return false;

}

errorMessage.style.display = 'none';

return true;

}

</script>
